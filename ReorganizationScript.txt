CREATE SEQUENCE AuditId MINVALUE 1 INCREMENT 1;
CREATE TABLE Audit (AuditId bigint NOT NULL DEFAULT nextval('AuditId'), AuditDate date NOT NULL , AuditTableName CHAR(100) NOT NULL , AuditDescription VARCHAR(200) NOT NULL , AuditShortDescription VARCHAR(200) NOT NULL , AuditAction VARCHAR(10) NOT NULL , SecUserId bigint NOT NULL , PRIMARY KEY(AuditId));
